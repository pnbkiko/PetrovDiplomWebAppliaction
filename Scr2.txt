-- Очистка существующих таблиц (если нужно)

DROP TABLE IF EXISTS maintenance_types CASCADE;

-- ============================================
-- 1. ТИПЫ ОБОРУДОВАНИЯ
-- ============================================
CREATE TABLE machine_types (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL UNIQUE,
    description TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ============================================
-- 2. ОБОРУДОВАНИЕ
-- ============================================
CREATE TABLE machines (
    id SERIAL PRIMARY KEY,
    type_id INTEGER NOT NULL REFERENCES machine_types(id) ON DELETE RESTRICT,
    model VARCHAR(255) NOT NULL,
    modification VARCHAR(255),
    manufacturer VARCHAR(255) NOT NULL,
    manufacturing_year INTEGER,
    serial_number VARCHAR(255) UNIQUE,
    inv_number VARCHAR(255) NOT NULL UNIQUE,
    commissioned_at DATE NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Индексы для быстрого поиска
CREATE INDEX idx_machines_type_id ON machines(type_id);
CREATE INDEX idx_machines_model ON machines(model);
CREATE INDEX idx_machines_manufacturer ON machines(manufacturer);
CREATE INDEX idx_machines_manufacturing_year ON machines(manufacturing_year);

-- ============================================
-- 3. ВИДЫ ТЕХНИЧЕСКОГО ОБСЛУЖИВАНИЯ
-- ============================================
CREATE TABLE maintenance_types (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    interval_days INTEGER NOT NULL CHECK (interval_days > 0)
);

-- ============================================
-- 4. ГРАФИК ТЕХНИЧЕСКОГО ОБСЛУЖИВАНИЯ
-- ============================================
CREATE TABLE maintenance_schedule (
    machine_id INTEGER NOT NULL REFERENCES machines(id) ON DELETE CASCADE,
    type_id INTEGER NOT NULL REFERENCES maintenance_types(id) ON DELETE CASCADE,
    next_due DATE NOT NULL,
    last_done DATE,
    PRIMARY KEY (machine_id, type_id)
);

CREATE INDEX idx_schedule_next_due ON maintenance_schedule(next_due);

-- ============================================
-- 5. АКТЫ ТЕХНИЧЕСКОГО ОБСЛУЖИВАНИЯ
-- ============================================
CREATE TABLE maintenance_acts (
    id SERIAL PRIMARY KEY,
    machine_id INTEGER NOT NULL REFERENCES machines(id) ON DELETE CASCADE,
    type_id INTEGER NOT NULL REFERENCES maintenance_types(id) ON DELETE CASCADE,
    date DATE NOT NULL,
    engineer VARCHAR(255) NOT NULL,
    notes TEXT,
    signed BOOLEAN DEFAULT FALSE
);

CREATE INDEX idx_acts_machine_id ON maintenance_acts(machine_id);
CREATE INDEX idx_acts_type_id ON maintenance_acts(type_id);
CREATE INDEX idx_acts_date ON maintenance_acts(date);

-- ============================================
-- 6. ТРИГГЕР ДЛЯ АВТОМАТИЧЕСКОГО ОБНОВЛЕНИЯ ГРАФИКА
-- ============================================
CREATE OR REPLACE FUNCTION update_next_due()
RETURNS TRIGGER AS $$
DECLARE
    v_interval_days INTEGER;
BEGIN
    -- Получаем интервал для данного типа ТО
    SELECT interval_days INTO v_interval_days 
    FROM maintenance_types 
    WHERE id = NEW.type_id;
    
    -- Проверяем, существует ли запись в графике
    IF EXISTS (SELECT 1 FROM maintenance_schedule WHERE machine_id = NEW.machine_id AND type_id = NEW.type_id) THEN
        -- Обновляем существующую запись
        UPDATE maintenance_schedule
        SET last_done = NEW.date,
            next_due = NEW.date + v_interval_days
        WHERE machine_id = NEW.machine_id AND type_id = NEW.type_id;
    ELSE
        -- Создаем новую запись в графике
        INSERT INTO maintenance_schedule (machine_id, type_id, last_done, next_due)
        VALUES (NEW.machine_id, NEW.type_id, NEW.date, NEW.date + v_interval_days);
    END IF;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Создаем триггер
DROP TRIGGER IF EXISTS trg_update_next_due ON maintenance_acts;
CREATE TRIGGER trg_update_next_due
AFTER INSERT ON maintenance_acts
FOR EACH ROW
EXECUTE FUNCTION update_next_due();

-- Проверка количества записей
SELECT 'Типы оборудования' as таблица, COUNT(*) as записей FROM machine_types
UNION ALL
SELECT 'Оборудование', COUNT(*) FROM machines
UNION ALL
SELECT 'Виды ТО', COUNT(*) FROM maintenance_types
UNION ALL
SELECT 'График ТО', COUNT(*) FROM maintenance_schedule
UNION ALL
SELECT 'Акты ТО', COUNT(*) FROM maintenance_acts
ORDER BY таблица;

-- Оборудование с полной информацией
SELECT 
    m.id,
    mt.name as тип,
    m.model as модель,
    m.modification as модификация,
    m.manufacturer as производитель,
    m.manufacturing_year as год_выпуска,
    m.serial_number as серийный_номер,
    m.inv_number as инв_номер,
    m.commissioned_at as дата_ввода
FROM machines m
JOIN machine_types mt ON m.type_id = mt.id
ORDER BY m.id;

-- Просроченные ТО
SELECT 
    m.model,
    m.modification,
    mt2.name as тип_ТО,
    ms.next_due as дата_выполнения,
    ms.next_due - CURRENT_DATE as дней_просрочки
FROM maintenance_schedule ms
JOIN machines m ON ms.machine_id = m.id
JOIN maintenance_types mt2 ON ms.type_id = mt2.id
WHERE ms.next_due < CURRENT_DATE
ORDER BY ms.next_due;
